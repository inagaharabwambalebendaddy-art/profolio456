        // INITIALIZE LUCIDE ICONS

        document.getElementById('year').textContent = new Date().getFullYear();

        // DARK / LIGHT MODE TOGGLE
        const themeToggleBtn = document.getElementById('themeToggle');
        
        if (localStorage.getItem('theme') === 'dark' || (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
            document.documentElement.classList.add('dark');
        } else {
            document.documentElement.classList.remove('dark');
        }

        themeToggleBtn.addEventListener('click', () => {
            if (document.documentElement.classList.contains('dark')) {
                document.documentElement.classList.remove('dark');
                localStorage.setItem('theme', 'light');
            } else {
                document.documentElement.classList.add('dark');
                localStorage.setItem('theme', 'dark');
            }
        });

        // MOBILE NAVIGATION MENU
        const mobileMenuBtn = document.getElementById('mobileMenuBtn');
        const mobileMenu = document.getElementById('mobileMenu');

            mobileMenu.classList.toggle('hidden');
        });

        // SEO SEARCH ENGINE TAB SWITCHER
        function switchSeoTab(tab) {
            const vGoogle = document.getElementById('seoViewGoogle');
            const vImages = document.getElementById('seoViewImages');
            const vBing = document.getElementById('seoViewBing');

            const btnG = document.getElementById('tabGoogle');
            const btnI = document.getElementById('tabImages');
            const btnB = document.getElementById('tabBing');

            // Hide all
            vGoogle.classList.add('hidden');
            vImages.classList.add('hidden');
            vBing.classList.add('hidden');

            // Reset buttons
            [btnG, btnI, btnB].forEach(b => {
                b.className = "px-3 py-1.5 rounded-md text-slate-600 dark:text-slate-400 hover:text-slate-900 dark:hover:text-white transition-all";
            });

            const activeBtnClass = "px-3 py-1.5 rounded-md bg-white dark:bg-slate-900 shadow text-slate-900 dark:text-white transition-all font-semibold";

            if (tab === 'google') {
                vGoogle.classList.remove('hidden');
                btnG.className = activeBtnClass;
            } else if (tab === 'images') {
                vImages.classList.remove('hidden');
                btnI.className = activeBtnClass;
            } else if (tab === 'bing') {
                vBing.classList.remove('hidden');
                btnB.className = activeBtnClass;
            }
        }

        // COPY TO CLIPBOARD FUNCTION
        function copyToClipboard(text) {
            const tempInput = document.createElement('input');
            tempInput.value = text;
            document.body.appendChild(tempInput);
            tempInput.select();
            document.execCommand('copy');
            document.body.removeChild(tempInput);

            const toast = document.getElementById('toast');
            toast.classList.remove('hidden');
            setTimeout(() => {
                toast.classList.add('hidden');
            }, 3000);
        }
